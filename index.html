/* C Implementation of Undo & Redo Using Two Stacks */

#include <stdio.h>
#include <stdbool.h>

#define MAX_SIZE 100

// 1. GAME STATE STRUCTURE
typedef struct {
    int playerX;
    int playerY;
    int health;
    bool keyCollected;
    bool doorOpen;
} GameState;

// 2. STACK STRUCTURE (LIFO: Last In, First Out)
typedef struct {
    GameState items[MAX_SIZE];
    int top; // -1 when empty
} Stack;

void initStack(Stack *s) {
    s->top = -1;
}

bool isEmpty(Stack *s) {
    return s->top == -1;
}

bool isFull(Stack *s) {
    return s->top == MAX_SIZE - 1;
}

// PUSH OPERATION: O(1)
bool push(Stack *s, GameState state) {
    if (isFull(s)) {
        printf("Stack Overflow!\n");
        return false;
    }
    s->items[++(s->top)] = state;
    return true;
}

// POP OPERATION: O(1)
GameState pop(Stack *s) {
    GameState empty = {0};
    if (isEmpty(s)) {
        printf("Stack Underflow!\n");
        return empty;
    }
    return s->items[(s->top)--];
}

// 3. UNDO OPERATION
// Pop previous state from Undo Stack, save current state to Redo Stack
bool undo(Stack *undoStack, Stack *redoStack, GameState *current) {
    if (isEmpty(undoStack)) {
        printf("Undo Stack is EMPTY!\n");
        return false;
    }
    // Step A: Push current state to Redo Stack
    push(redoStack, *current);

    // Step B: Pop previous state from Undo Stack and restore
    *current = pop(undoStack);
    return true;
}

// 4. REDO OPERATION
// Pop state from Redo Stack, save current state to Undo Stack
bool redo(Stack *undoStack, Stack *redoStack, GameState *current) {
    if (isEmpty(redoStack)) {
        printf("Redo Stack is EMPTY!\n");
        return false;
    }
    // Step A: Push current state to Undo Stack
    push(undoStack, *current);

    // Step B: Pop forward state from Redo Stack and restore
    *current = pop(redoStack);
    return true;
}

// 5. NEW MOVEMENT (Crucial DSA Stack Rule)
// Pushes current state to Undo Stack and CLEARS Redo Stack
void onNewMove(Stack *undoStack, Stack *redoStack, GameState *current, GameState nextState) {
    // Step A: Push current state before move
    push(undoStack, *current);

    // Step B: Divergent path invalidates redo history -> CLEAR REDO STACK
    redoStack->top = -1;

    // Step C: Apply new move
    *current = nextState;
}
